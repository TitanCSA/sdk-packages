cmake_minimum_required(VERSION 3.24)
project(NodeInterceptorSDK VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 required")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(TitanNodeInterceptor STATIC sdk/Bridge.cpp sdk/SDK/Interceptor.cpp)
add_library(NodeInterceptorSDK::sdk ALIAS TitanNodeInterceptor)
set_target_properties(TitanNodeInterceptor PROPERTIES EXPORT_NAME sdk)
target_compile_features(TitanNodeInterceptor PUBLIC cxx_std_17)
target_compile_definitions(TitanNodeInterceptor PUBLIC INTERCEPTION_STATIC PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(TitanNodeInterceptor PRIVATE /utf-8 /EHsc /W4)
target_include_directories(TitanNodeInterceptor PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/sdk>" "$<INSTALL_INTERFACE:include>")
target_link_libraries(TitanNodeInterceptor PUBLIC setupapi cfgmgr32 hid advapi32 bcrypt)
install(TARGETS TitanNodeInterceptor EXPORT NodeInterceptorSDKTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY sdk/ DESTINATION include FILES_MATCHING PATTERN "*.h")
install(EXPORT NodeInterceptorSDKTargets NAMESPACE NodeInterceptorSDK:: DESTINATION lib/cmake/NodeInterceptorSDK)
configure_file(NodeInterceptorSDKConfig.cmake.in NodeInterceptorSDKConfig.cmake @ONLY)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/NodeInterceptorSDKConfigVersion.cmake" VERSION 1.0.0 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/NodeInterceptorSDKConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/NodeInterceptorSDKConfigVersion.cmake" DESTINATION lib/cmake/NodeInterceptorSDK)
install(FILES LICENSE NOTICE DESTINATION share/node-interceptor-sdk)
install(DIRECTORY licenses/ DESTINATION share/node-interceptor-sdk/licenses OPTIONAL)
