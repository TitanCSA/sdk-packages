cmake_minimum_required(VERSION 3.24)
project(node_interceptor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(NODE_HEADERS "" CACHE PATH "Node.js headers directory (contains node_api.h)")
set(NODE_LIBRARY "" CACHE FILEPATH "Matching Windows node.lib")
option(INTERCEPTOR_TEST_BACKEND "Use the synthetic SDK, never a physical driver" OFF)
add_library(node_interceptor MODULE src/binding.cc src/DelayLoad.cpp native/Bridge.cpp)
set_target_properties(node_interceptor PROPERTIES PREFIX "" SUFFIX ".node")
target_include_directories(node_interceptor PRIVATE native node_modules/node-addon-api "${NODE_HEADERS}")
target_compile_definitions(node_interceptor PRIVATE NAPI_VERSION=8 INTERCEPTION_STATIC UNICODE _UNICODE NOMINMAX)
target_compile_options(node_interceptor PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(node_interceptor PRIVATE "${NODE_LIBRARY}" delayimp)
target_link_options(node_interceptor PRIVATE /DELAYLOAD:node.exe)
if(INTERCEPTOR_TEST_BACKEND)
  target_sources(node_interceptor PRIVATE test/native/FakeSdk.cpp)
  target_compile_definitions(node_interceptor PRIVATE INTERCEPTOR_TEST_BACKEND)
else()
  target_sources(node_interceptor PRIVATE native/SDK/Interceptor.cpp)
  target_link_libraries(node_interceptor PRIVATE setupapi cfgmgr32 hid advapi32 bcrypt)
endif()
