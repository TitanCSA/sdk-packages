cmake_minimum_required(VERSION 3.24)
project(BrowserControlSDKJsonSDK LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PRODUCT_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../packaging/native" CACHE PATH "Canonical SDK source")
if(NOT EXISTS "${PRODUCT_NATIVE_SOURCE}/CMakeLists.txt")
 set(PRODUCT_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../Browser-Control-SDK-CPP-vcpkg")
endif()
add_subdirectory("${PRODUCT_NATIVE_SOURCE}" native)
add_library(BrowserControlSDKJsonSDK SHARED Bridge.cpp)
target_include_directories(BrowserControlSDKJsonSDK PRIVATE "${PRODUCT_NATIVE_SOURCE}/sdk")
target_compile_features(BrowserControlSDKJsonSDK PRIVATE cxx_std_20)
target_compile_options(BrowserControlSDKJsonSDK PRIVATE /utf-8 /EHsc /W4)
target_compile_definitions(BrowserControlSDKJsonSDK PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(BrowserControlSDKJsonSDK PRIVATE BrowserControlSDK::sdk)
