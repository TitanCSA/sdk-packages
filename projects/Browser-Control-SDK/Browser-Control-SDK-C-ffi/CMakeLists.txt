cmake_minimum_required(VERSION 3.24)
project(BrowserControlFFI VERSION 0.2.2 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BCS_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../packaging/native" CACHE PATH "Canonical Browser SDK source")
if(NOT EXISTS "${BCS_NATIVE_SOURCE}/CMakeLists.txt")
  set(BCS_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../Browser-Control-SDK-CPP-vcpkg")
endif()
add_subdirectory("${BCS_NATIVE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/native")
add_library(BrowserControlBridge SHARED BrowserControlBridge.cpp)
target_link_libraries(BrowserControlBridge PRIVATE BrowserControlSDK::sdk)
target_compile_features(BrowserControlBridge PRIVATE cxx_std_17)
target_compile_definitions(BrowserControlBridge PRIVATE BCS_BUILD WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(BrowserControlBridge PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>")
option(BCS_BUILD_TESTS "Build the inert C ABI consumer" OFF)
if(BCS_BUILD_TESTS)
  enable_testing()
  add_executable(browser_c_consumer tests/c_consumer.c)
  target_link_libraries(browser_c_consumer PRIVATE BrowserControlBridge)
  add_test(NAME browser_c_consumer COMMAND browser_c_consumer)
endif()
