cmake_minimum_required(VERSION 3.24)
project(InterceptorMultiLang VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(ICB_TEST_BACKEND "Synthetic backend; never opens a physical driver" OFF)
option(ICB_SANITIZE "AddressSanitizer for native fixture checks" OFF)
set(NODE_HEADERS "" CACHE PATH "Folder containing node_api.h")
set(NODE_LIBRARY "" CACHE FILEPATH "Matching Windows Node import library")
add_library(InterceptorBridge SHARED native/Bridge.cpp)
target_compile_features(InterceptorBridge PRIVATE cxx_std_17)
target_compile_definitions(InterceptorBridge PRIVATE IC_BRIDGE_EXPORTS UNICODE _UNICODE NOMINMAX)
target_compile_options(InterceptorBridge PRIVATE /W4 /WX /EHsc /utf-8)
target_include_directories(InterceptorBridge PUBLIC native PRIVATE sdk)
if(ICB_SANITIZE)
  target_compile_options(InterceptorBridge PRIVATE /fsanitize=address /Zi)
  target_link_options(InterceptorBridge PRIVATE /INCREMENTAL:NO /DEBUG)
endif()
if(ICB_TEST_BACKEND)
  target_sources(InterceptorBridge PRIVATE tests/FakeSdk.cpp)
else()
  target_sources(InterceptorBridge PRIVATE sdk/Interceptor.cpp)
  target_link_libraries(InterceptorBridge PRIVATE setupapi cfgmgr32 hid advapi32 bcrypt ole32)
endif()
if(NODE_HEADERS AND NODE_LIBRARY)
  add_library(interceptor_node MODULE bindings/node/native.cpp bindings/node/DelayLoad.cpp)
  set_target_properties(interceptor_node PROPERTIES PREFIX "" SUFFIX ".node")
  target_include_directories(interceptor_node PRIVATE native "${NODE_HEADERS}")
  target_compile_features(interceptor_node PRIVATE cxx_std_17)
  target_compile_definitions(interceptor_node PRIVATE NAPI_VERSION=8 NODE_GYP_MODULE_NAME=interceptor_node)
  target_link_libraries(interceptor_node PRIVATE InterceptorBridge "${NODE_LIBRARY}" delayimp)
  target_link_options(interceptor_node PRIVATE /DELAYLOAD:node.exe)
  if(ICB_TEST_BACKEND)
    target_compile_definitions(interceptor_node PRIVATE ICB_TEST_BACKEND)
  endif()
endif()
add_executable(c_example examples/c/main.c)
target_link_libraries(c_example PRIVATE InterceptorBridge)
if(NOT ICB_TEST_BACKEND)
  add_executable(cpp_example examples/cpp/main.cpp sdk/Interceptor.cpp)
  target_include_directories(cpp_example PRIVATE sdk)
  target_compile_features(cpp_example PRIVATE cxx_std_17)
  target_compile_definitions(cpp_example PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(cpp_example PRIVATE setupapi cfgmgr32 hid advapi32 bcrypt ole32)
endif()
include(CTest)
if(BUILD_TESTING AND ICB_TEST_BACKEND)
  add_executable(bridge_tests tests/BridgeTests.cpp)
  target_link_libraries(bridge_tests PRIVATE InterceptorBridge)
  add_test(NAME bridge_contract COMMAND bridge_tests)
endif()
